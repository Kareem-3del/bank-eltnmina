أبرز الاتفاقيات ومذكرات التعاون التي أبرمها الصندوق

واصــل صنــدوق التنميــة العقاريــة خلال عــام 2025م توســيع شــبكة شراكاتــه مــع الجهــات الحكوميــة والقطــاع المــالي والاســتثماري والتــقني، عبر توقيـــــع حزمــــــة متكاملــة مــن الاتفاقيــات ومذكــرات التفاهــم التي غطــت مختلــف حلقــات سلســلة القيمــة الإســكانية. وشــملت هــذه الشــراكات تطويــر البرامــج التمويليــة والدعـــــم الإضافــي، وتمويــل مشــاريع البيـع على الخارطـة، وتعزيز الاسـتدامة المالية، وتفعيــل الحلــول الرقميـة، وتبـادل وتحليل البيانــات، ورفــع كفـاءة التحصيـل والعمليـات، إضافـة إلى تنميــة القــدرات المؤسســية والاســتثمارية.

معلومات عن الأداء المالي للصندوق

حافـظ الصنـدوق العقـاري حتى نهايـة عــام 2025م على قاعــدة رأســمالية قويــة، حيــث بلــغ رأس المــال الــمصرح بــه نحــو 190.9 مليــار ، وبلــغ رأس المــال المدفــوع نحــو 180.2 مليــار

الاستدامة المالية

ينطلـــق صنـــدوق التنميـــة العقاريـــة الاستدامة المالية فــي تحقيــق الاســتدامة الماليــة مــن منظومـــة تمويليـــة متجـــددة تقـــوم على تنــــويـــع مصـــادر الدخـــل، وابتـــكار فـــرص اســـتثمارية عقاريـــة، وتعظيـــم كفـــاءة اســـتثمار المـــوارد، إلى جانـــب عقـــد شراكات مـــع القطـــاع الخـــاص، وتنميــة محفظــة الاســتثمارات ضمــن إطـــار إدارة المخاطـــر، وتحـــسين كفـــاءة الإنفـــاق التشـــغيلي، بمـــا يضمـــن اســـتدامة دوره فـــي تمـــكين الأسر مـــن تملـــك المســـكن.

من أهم الأعمال التي نّفذها الصندوق في مجال الاستثمار العقاري ما يلي:

  • 1

    خلــق فــرص اســتثماريـة مــع الشركات الماليــــة فـي مختلـــف مناطــــق المملكــــة، حيــث بلــــغ إجمــــالي حجــــــم الصناديـــــق 1.21 مليــــار ســعودي. ومــن المتوقــع أن تســهم الصناديــق فــي تطويــــر وتــوفيـــــر أكثر مــن 1,500 وحــدة ســكنية، بمــا يدعــم نمــو المعــروض الســكـني واســتدامته.

  • 2

    تــم إطلاق منتــج سلاســل الإمــداد خــلال الربــع الأول مــن عــام 2025م، والذي يهـدف إلى دعـم المطورين العقـاريين مـن خلال سـداد مسـتحقات المواد والمستلزمات الانشائية من قبل الموردين.

  • 3

    تم توقيـع اتفاقية للاسـتحواذ على الشركة الوطنية المســانــدة لخدمــات التمويــل، وذلــك مــن خلال شراء حصــة أســهم الشركــة الوطنيــة للإســــكان فــي الشركــة، بمــا يدعــم التكامــل المـؤسســي وتعزيـز منظومـة التمويل.

  • 4

    تــم توقيــع اتفاقيــة لإدارة محفظــة اســتثمارية مــع شركـــة جــدوى للاسـتثمـار، بهـدف توجيـــه الاسـتثمــارات نحــو الصناديـــــق الاسـتثماريـــــة العقاريــة ذات الجــدوى العاليــة.

  • 5

    التواصــل والعمـل مــع 10 شركات ماليــة، لبحث واســتعــراض الفــرص التمويليـة والاسـتثماريـــة العقاريـة فـي الســوق المــحلي.

  • 6

    يجري العمــل حاليـــًا مــع مــدراء أصــول خارجييــن للاســتثمار فــي الصناديــق العقاريــة.

  • 7

    تــم العمــل بالتعــاون مــع وزارة الإســكان على إصــدار مــؤشرات للتمويــل العقـــاري للصنــدوق للــنشرة الاقتصاديــــة مــن خــلال منصــة بلــدي.

  • 8

    توقيع مذكرات تفاهـم مــع عــدة جهــاتٍ ماليــة في معـــرض سـيتي سكيــــب 2025م، لتعزيـــز التعـاوِن في الاستثمارات العقاريــة وزيـادة مجال الفـرص المتاحــة.

  • 9

    دعــم شركـــات التمويــــل العقـــاري مــن خــــلال تــوفير السيولةِ المدعومةِ الــتي تهــدف إلى دعـم تكلفــــة التمويــل العقــاري للأفـــــراد بإجمــــالي بلـغ 200 مليـون ، مـا نتـج عنـه إنشـاء عقــــاري بمتوسـط 344 عقــد تمويـلٍ مبلــغ 581 ألــف لــكلَّ عقــد.

  • 10

    قيــام برنامــج تمـكين المطوريــن العقاريين بدعم شركـات التطويـــــر العقــــاري حيــث تــم اعتمـــاد تسـهيـــــلات بلــغ إجماليهــا 958,375 مليــون لإنشـــاء +35 مشروعًا عقاريًّا بإجمـــالي زيــادة 4,586 وحــدًة ســكنيًة، بما يعــِّز المعــروض مـن المنتجــات الســكنيـــة فــي الســــوق ويــــدعــم مســتهدفات التملــك.

العائد على الاستثمار

حققـــت المحفظـــة الاستثمارية عائـــدًا بمعـــدل %5.11 خلال عـــام 2025م

الإيرادات التشغيلية (مليون)

وقد تجاوزت الإيــرادات التشــغيلية المستهدف السنوي لتصـل إلى نحـو 949.3 مليون مقابل مســتهدف يقــارب 792.7 مليون بنســبة تحقـق بلغـت %120

من أهم الأعمال التي حققها قطاع الخزينة خلال عام 2025م:

  • مشــــروع تــحــديـــــــث سيـــــاســـــة الاسـتـثـمـــــــار.
  • توقيع اتفاقيات إدارة محافظ استثمار عقارية ومحافظ خاصة للأسهم المحليــة.
  • تعــــزيــز التخصيــص فــي الاســتثمــــارات الاستراتيجيــــة طويلــــــة الأجـــــــل.
  • توزيع المخاطر الاستثمارية من خلال زيادة التخصيص في سوق الأسهم المحلية.

أداء الاستثمارات:

حققـــت المحفظـــة الاســـتثمارية عائـــدًا بمعـــدل %5.11 خلال عـــام 2025م مقارنـــة بــ %5.42 لعـــام 2024م حيــث انخفــض العائــد تماشــيًا مـــع الانخفـــاض فـــي أســـعار الفائـــدة للودائــع والصكــوك الاســتثمارية فــي الأســـواق الماليـــة.

التحصيل

فـــي إطـــار تعزيـــز الاســـتدامــــة الماليـــة لصنـــدوق التنميـــــــة العقاريـــة، جـــرى التركيز خـــــلال عـــام 2025م على رفـــع كفــــــاءة تحصيـــل محفظـــة القـــروض المبـــاشرة بوصفهـــا إحـــدى الركائـــز الاستراتيجيـــة للأداء المــالي. وانطلاقـــــــــًا مـــن ذلـــك، تـــم تطويـــر الخدمـــات المقدمـــة للمســـتفيدين، وتحديـــث قواعـــد البيانـــات، وأتمتـــة عمليـــات التحصيـــل، إلى جانـــب تـــوفير قنــــوات دفـــع جديـــدة، وتكثيـــف الحـــملات التحصيليـــة، وإطـــلاق برنامـــج خصـــم الســداد المبكــر. وقــد أســهمت هــذه الجهـــود مجتمعـــةً فـي رفـع نســبة التحصيل، وخفـــض معـــدلات التعثر، وتحقيـــق المسـتهدفات المحددة،
